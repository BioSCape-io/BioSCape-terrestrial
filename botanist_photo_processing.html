Parse and combine all botanist sheets

Each botanist worksheet was read and combined into a single dataset. Rows lacking a timestamp were discarded as incomplete entries. The modify_date field (formatted YYYY:MM:DD HH:MM:SS by the camera firmware) was re-formatted to ISO-8601 and parsed into separate datetime, date, and time columns. GPS coordinate fields were coerced to numeric; cells containing non-numeric text from Excel were silently converted to NA. Photos were classified as either "plot" (cardinal direction shots, plot overview, or rarefaction sequence markers) or "species" (individual taxon records) based on the content of the genus field. Cardinal directions were matched as exact single-character strings to avoid false positives within genus names. Records with missing GPS coordinates (e.g. screenshots) were excluded, and the resulting table was converted to an sf point layer in WGS 84 (EPSG:4326).

Show the code 
sheets = excel_sheets(fname)
# Read sheets and do a little cleaning for irregularites

sheets=sheets[-1] # Drop the old ross spreadsheet

photo_all=foreach(i=sheets,.combine=bind_rows) %do% {
  read_xlsx(fname,i) |>
    # Drop rows with no timestamp — these are incomplete entries and cannot be
    # placed on a timeline; also prevents unparseable strings downstream.
    filter(!is.na(modify_date)) |>
    mutate(
      # modify_date arrives as a character from Excel: "YYYY:MM:DD HH:MM:SS"
      # Replace the colons in the date portion with dashes before parsing.
      fdate    = gsub(":", "-", substring(modify_date, 1, 10)),
      fdate2   = paste(fdate, substring(modify_date, 12)),
      datetime = as_datetime(fdate2),
      date     = date(datetime),
      time     = hms(format(datetime, "%H:%M:%S")),
      # GPS fields can arrive as character "NA" from Excel; suppress coercion
      # warnings because NA → NA is the intended behaviour here.
      gps_latitude      = suppressWarnings(as.numeric(gps_latitude)),
      gps_longitude     = suppressWarnings(as.numeric(gps_longitude)),
      gps_altitude      = suppressWarnings(as.numeric(gps_altitude)),
      gps_position_error = suppressWarnings(as.numeric(gps_position_error)),
      modify_date  = as_datetime(modify_date),
      plot_number  = as.character(plot_number),
      # Classify plot-level vs species photos based on the genus field.
      # Cardinal directions (S, N, W, E) are matched as whole words only to avoid
      # false positives inside genus names (e.g. "Erica" contains "E").
      photo_type = case_when(
        grepl("^(plot|start|rarefaction|end|stop|centre|center|magnet|cairn)$", genus, ignore.case = TRUE) ~ "plot",
        grepl("^[SNWE]$", genus)                                                 ~ "plot",
        TRUE                                                                     ~ "species"
      )
    ) |>
    select(-fdate, -fdate2) |>       # drop temporary date-parsing columns
    select(-exposure_time) |>        # drop — inconsistent type across sheets
    filter(!is.na(gps_longitude) & !is.na(gps_latitude)) |>  # drop screenshots with no GPS
    st_as_sf(coords = c("gps_longitude", "gps_latitude"), remove = FALSE,
             crs = 4326)
}

Join with plot-level species lists

The vegetation survey species list records species actually observed and identified during structured vegetation surveys at each plot. Survey identifications are treated as the authoritative name reference. Photo-recorded names are corrected against this list using Jaro-Winkler fuzzy string matching before any cross-dataset comparison is performed, so that minor field-recording spelling errors do not inflate mismatch counts.

Fuzzy-correct photo names against the verified survey list

Each distinct photo taxon name is matched to the closest name in the verified survey list using the Jaro-Winkler string distance metric, which weights character differences near the start of a string more heavily — a good fit for binomial taxon names where genus misspellings are common. Photo names within the distance threshold receive the nearest matching spd name as a correction; those above threshold are flagged as "no_match" and retain their original values. Corrections are made at the name level (not per plot), because a misspelling is a property of the entered string, not of the plot where it was recorded.

Show the code 
# Jaro-Winkler distance threshold: photo names with a best-match distance below
# this value are corrected to the spd name; those above are flagged as unverified.
jw_threshold <- 0.10

# Distinct taxon names entered in the photo dataset (species-type photos only).
photo_names_distinct <- photo_all2 |>
  st_drop_geometry() |>
  filter(photo_type == "species", !is.na(genus), !is.na(plot)) |>
  mutate(photo_name = paste(str_trim(genus), str_trim(species))) |>
  distinct(photo_name)

# Compute a full n_photo × n_spd Jaro-Winkler distance matrix, then extract the
# single best-matching spd name for each photo name.
dist_matrix <- stringdistmatrix(
  str_to_lower(photo_names_distinct$photo_name),
  str_to_lower(spd_names_global),
  method = "jw", p = 0.1   # p = 0.1 is the standard Jaro-Winkler prefix weight
)

best_idx  <- apply(dist_matrix, 1, which.min)
best_dist <- apply(dist_matrix, 1, min)

photo_name_corrections <- photo_names_distinct |>
  mutate(
    spd_name_matched  = spd_names_global[best_idx],
    match_dist        = round(best_dist, 4),
    match_type        = case_when(
      str_to_lower(photo_name) == str_to_lower(spd_name_matched) ~ "exact",
      match_dist < jw_threshold                                   ~ "fuzzy",
      TRUE                                                        ~ "no_match"
    ),
    # Use the verified spd name when a sufficiently close match exists
    corrected_name    = if_else(match_type != "no_match",
                                spd_name_matched, photo_name),
    corrected_genus   = word(corrected_name, 1),
    corrected_species = word(corrected_name, 2)
  )

Apply corrections to the photo dataset

The correction lookup is joined back to photo_all2, adding corrected_genus, corrected_species, match_type, and match_dist columns. The corrected names are used in all subsequent joins; the original genus / species columns are preserved unmodified for traceability.

Show the code 
# Build a temporary join key in photo_all2, join the corrections, then drop the key.
photo_all2 <- photo_all2 |>
  mutate(
    photo_name = if_else(
      photo_type == "species" & !is.na(genus) & !is.na(plot),
      paste(str_trim(genus), str_trim(species)),
      NA_character_
    )
  ) |>
  left_join(
    select(photo_name_corrections,
           photo_name, corrected_genus, corrected_species, match_type, match_dist),
    by = "photo_name"
  ) |>
  # For non-species photos the join key is NA; fall back to original values
  mutate(
    corrected_genus   = coalesce(corrected_genus,   genus),
    corrected_species = coalesce(corrected_species, species)
  ) |>
  select(-photo_name)

# Per-plot species list using corrected names — used for all subsequent joins.
photo_species <- photo_all2 |>
  st_drop_geometry() |>
  filter(photo_type == "species", !is.na(corrected_genus), !is.na(plot)) |>
  mutate(
    genus_l   = str_to_lower(str_trim(corrected_genus)),
    species_l = str_to_lower(str_trim(corrected_species))
  ) |>
  distinct(plot,
           genus   = corrected_genus,
           species = corrected_species,
           genus_l, species_l)

Exploratory Summary

plot-level summary

Per-plot summary statistics were computed, including counts of photos by type (species, rarefaction, plot overview, iNaturalist-flagged), the number of unique taxa photographed, rarefaction effort, survey dates, and GPS quality indicators. The summary table was written to disk for use in downstream analyses.

Show the code 
plot_summary <- photo_all2 |>
  st_drop_geometry() |>
  # Exclude photos with no matched plot before summarising
  filter(!is.na(plot)) |>
  group_by(plot) |>
  summarise(
    # Botanists who contributed photos to this plot
    botanist = paste(sort(unique(botanist)), collapse = ", "),

    # Total photo counts
    photo_total        = n(),
    photo_species      = sum(photo_type == "species", na.rm = TRUE),
    photo_rarefaction  = sum(rarefaction_photo == 1, na.rm = TRUE),
    photo_plot         = sum(photo_type == "plot", na.rm = TRUE),
    # Photos flagged for iNaturalist upload
    photo_inat         = sum(inat_photo == TRUE, na.rm = TRUE),
    # Rarefaction duplicates (same individual re-photographed in sequence)
    photo_rarefaction_duplicate = sum(
      rarefaction_duplicate == TRUE, na.rm = TRUE
    ),

    # Species richness: distinct genus+species combos from species-type photos,
    # excluding rows where genus is NA (unidentified / plot-level shots)
    n_species = n_distinct(paste(
      genus[photo_type == "species" & !is.na(genus)],
      species[photo_type == "species" & !is.na(genus)]
    )),

    # Maximum rarefaction replicate number — indicates depth of rarefaction
    # sequence completed for this plot
    n_rarefaction_replicates = sum(rarefaction_replicate, na.rm = TRUE),

    # Survey effort: number of unique dates photos were taken
    n_dates    = n_distinct(date),
    date_first = min(date, na.rm = TRUE),
    date_last  = max(date, na.rm = TRUE),

    # Spatial QC: median distance from photos to the official plot centroid (m)
    median_dist_to_plot_m = median(dist_to_plot_m, na.rm = TRUE),
    # GPS precision reported by the device
    mean_gps_error_m = round(mean(gps_position_error, na.rm = TRUE), 1),
    .groups = "drop"
  ) |>
  # Replace -Inf / Inf that arise from max/min on all-NA groups
  mutate(
    across(where(is.numeric), \(x) ifelse(is.infinite(x), NA_real_, x))
  )

Rename and organize all photos

Photos were copied from the shared Google Drive source folders into a standardised local directory structure (data/photo_out/<BScpPID>/), where unmatched photos (no GPS-derived plot assignment) are placed in a 00/ subfolder. Each file was renamed according to the convention <BScpPID>_<photo_type>_<YYYYMMDD_HHMMSS>_<botanist>.jpeg, where the botanist suffix disambiguates the rare case of two photographers capturing a photo at the same plot within the same second. Files are copied byte-for-byte with file.copy() to preserve all original EXIF metadata intact; exiftool is then used to add structured provenance fields on top without re-encoding the image. This chunk is set to eval=F and must be run manually after verifying the Google Drive paths.

Show the code 
photo_all3 <- photo_all2 |>
  st_drop_geometry() |>
  mutate(
    # Use "00" as the plot folder for photos that could not be matched to a plot.
    plot_folder = if_else(is.na(BScpPID), "00", BScpPID),
    # Sanitise botanist name for use in a filename (replace spaces / special chars
    # with underscores).
    botanist_safe = str_replace_all(botanist, "[^A-Za-z0-9]", "_"),
    # Naming convention: <plot>_<type>_<YYYYMMDD_HHMMSS>_<botanist>.jpeg
    # Appending botanist prevents silent overwrites when two photographers take a
    # photo at the same plot within the same second.
    newfile = paste0(
      plot_folder, "/",
      plot_folder, "_", photo_type, "_",
      format(datetime, "%Y%m%d_%H%M%S"), "_",
      botanist_safe, ".jpeg"
    )
  )

output_folder <- "data/photo_out/"

purrr::walk(seq_len(nrow(photo_all3)), function(i) {

  input_file  <- file.path(
    "data/photos/Photos",
    sub("BioSCape_Admin/VegPlots/Photos/", "", photo_all3$folder[i]),
    photo_all3$filename[i]
  )
  output_file <- file.path(output_folder, photo_all3$newfile[i])

  if (!dir.exists(dirname(output_file)))
    dir.create(dirname(output_file), recursive = TRUE)

  input_ext <- str_to_lower(tools::file_ext(input_file))

  if (input_ext %in% c("heic", "heif")) {
    # HEIC/HEIF must be converted to JPEG — re-encoding is unavoidable.
    # After writing the pixel data, exiftool copies all original metadata
    # (GPS, camera model, exposure, etc.) from the source HEIC back into the
    # new JPEG so that provenance is preserved as completely as possible.
    image_read(input_file) |>
      image_write(output_file, format = "jpeg", quality = 95)
    # Copy original EXIF/XMP/IPTC from the HEIC source into the new JPEG.
    # The source path is wrapped in shQuote() because exiftool_call does not
    # shell-quote elements of the args list (only fnames is quoted), and paths
    # with spaces would otherwise be split by the shell into multiple tokens.
    exiftool_call(
      args = list(
        "-overwrite_original",
        paste0("-TagsFromFile=", shQuote(input_file)),
        "-all:all"
      ),
      fnames = output_file
    )
  } else {
    # For native JPEGs: copy byte-for-byte so ALL original EXIF/IPTC/XMP
    # metadata (GPS, camera model, exposure settings, colour profile, etc.)
    # is preserved without any re-encoding loss.
    file.copy(input_file, output_file, overwrite = TRUE)
  }

  # Build the corrected binomial for the Subject tag; fall back to the original
  # genus/species for plot-type photos where corrected_genus is not meaningful.
  taxon_subject <- if_else(
    photo_all3$photo_type[i] == "species" &
      !is.na(photo_all3$corrected_genus[i]),
    paste(photo_all3$corrected_genus[i], photo_all3$corrected_species[i]),
    ""
  )

  # Use the matched plot ID for the title, falling back to "unmatched" for the
  # "00" folder photos so the Title tag is still human-readable.
  plot_label <- if_else(
    is.na(photo_all3$BScpPID[i]), "unmatched", photo_all3$BScpPID[i]
  )

  # Sanitise free-text fields that may contain characters that would break the
  # exiftool command string (double-quotes, backslashes, newlines).
  sanitise <- function(x) str_replace_all(x, '["\\\\\n\r]', " ")

  # Each field is written as a separate element so that field values containing
  # spaces do not need additional quoting (exiftool treats each list element as
  # one argument).
  cmd <- list(
    "-overwrite_original",

    # XMP / IPTC provenance fields — structured, machine-readable ----------------
    # Original source path so the renamed copy is always traceable.
    paste0("-XMP:Source=",
           sanitise(paste0(photo_all3$folder[i], "/", photo_all3$filename[i]))),
    # Plot identifier as a dedicated XMP field for easy downstream filtering.
    paste0("-XMP:PlotID=", plot_label),
    # Botanist name as the EXIF Artist field (standard) and separately in XMP.
    paste0("-EXIF:Artist=",        sanitise(photo_all3$botanist[i])),
    paste0("-XMP:Creator=",        sanitise(photo_all3$botanist[i])),
    # Free-text description recorded in the field spreadsheet.
    paste0("-EXIF:ImageDescription=",
           sanitise(photo_all3$description[i])),
    # Species identification — use corrected name for Subject/Taxon tags.
    paste0("-IPTC:ObjectName=",    sanitise(taxon_subject)),
    paste0("-XMP:Subject=",        sanitise(taxon_subject)),
    # Photo type (species / plot) for easy filtering.
    paste0("-XMP:Type=",           photo_all3$photo_type[i]),
    # Title with plot and photo-type context.
    paste0("-EXIF:XPTitle=",
           sanitise(paste0("BioSCape ", photo_all3$photo_type[i],
                           " photo from plot ", plot_label))),
    # GPS coordinates from the spreadsheet as a cross-check against device GPS.
    paste0("-XMP:GPSLatitude=",    photo_all3$gps_latitude[i]),
    paste0("-XMP:GPSLongitude=",   photo_all3$gps_longitude[i]),
    # Copyright statement for the project.
    "-EXIF:Copyright=BioSCape project",
    "-XMP:Rights=BioSCape project",
    # Keywords for general discovery.
    paste0("-IPTC:Keywords=BioSCape"),
    paste0("-IPTC:Keywords=", sanitise(photo_all3$botanist[i])),
    paste0("-IPTC:Keywords=", plot_label)
  )

  exiftool_call(args = cmd, fnames = output_file)
})

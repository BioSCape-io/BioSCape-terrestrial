Parse and combine all botanist sheets

Each botanist worksheet was read and combined into a single dataset. Rows lacking a timestamp were discarded as incomplete entries. The modify_date field (formatted YYYY:MM:DD HH:MM:SS by the camera firmware) was re-formatted to ISO-8601 and parsed into separate datetime, date, and time columns. GPS coordinate fields were coerced to numeric; cells containing non-numeric text from Excel were silently converted to NA. Photos were classified as either "plot" (cardinal direction shots, plot overview, or rarefaction sequence markers) or "species" (individual taxon records) based on the content of the genus field. Cardinal directions were matched as exact single-character strings to avoid false positives within genus names. Records with missing GPS coordinates (e.g. screenshots) were excluded, and the resulting table was converted to an sf point layer in WGS 84 (EPSG:4326).

Show the code 
sheets = excel_sheets(fname)
# Read sheets and do a little cleaning for irregularites

sheets <- sheets[-1]  # Drop the old Ross spreadsheet (historical; pre-dates unified workflow)

# Read all botanist sheets and combine into one sf object.
# purrr::map() iterates over sheet names; bind_rows() stacks the results.
photo_all <- map(sheets, \(i)
  read_xlsx(fname, i) |>
    # Drop rows with no timestamp — incomplete entries that cannot be placed on
    # a timeline and would produce unparseable strings downstream.
    filter(!is.na(modify_date)) |>
    mutate(
      # Camera firmware writes dates as "YYYY:MM:DD HH:MM:SS"; replace the
      # date-portion colons with dashes so lubridate can parse the string.
      datetime    = ymd_hms(str_replace(modify_date,
                              "^(\\d{4}):(\\d{2}):(\\d{2})", "\\1-\\2-\\3")),
      date        = date(datetime),
      time        = hms(format(datetime, "%H:%M:%S")),
      modify_date = datetime,
      # GPS fields can arrive as character "NA" from Excel; suppress coercion
      # warnings because NA → NA is the intended behaviour here.
      gps_latitude       = suppressWarnings(as.numeric(gps_latitude)),
      gps_longitude      = suppressWarnings(as.numeric(gps_longitude)),
      gps_altitude       = suppressWarnings(as.numeric(gps_altitude)),
      gps_position_error = suppressWarnings(as.numeric(gps_position_error)),
      plot_number        = as.character(plot_number),
      # Classify plot-level vs species photos based on the genus field.
      # Cardinal directions (S, N, W, E) are matched as whole words to avoid
      # false positives inside genus names (e.g. "Erica" contains "E").
      # W2/S2/E2/N2 are sub-cardinal directions used by some botanists.
      photo_type = case_when(
        grepl("^(plot|start|rarefaction|end|stop|centre|center|magnet|cairn|view|datasheet|webbing|alluvial|unidentified)$",
              genus, ignore.case = TRUE)             ~ "plot",
        grepl("^[SNWE]$|W2|S2|E2|N2|South", genus)  ~ "plot",
        TRUE                                         ~ "species"
      )
    ) |>
    select(-exposure_time) |>   # drop — inconsistent column type across sheets
    filter(!is.na(gps_longitude) & !is.na(gps_latitude)) |>  # drop screenshots
    st_as_sf(coords = c("gps_longitude", "gps_latitude"), remove = FALSE,
             crs = 4326)
) |>
  bind_rows()

Assign photos to vegetation plots

BioSCape vegetation plot centroids and homogeneous-patch polygons were loaded from the versioned spatial products file. The Euclidean distance from each photo to every plot centroid was computed in the projected coordinate system of the plot layer. Each photo was assigned to its nearest plot; assignments were nullified for photos located more than 1,000 m from the nearest centroid, as these are unlikely to represent intended plot observations.

Clean species field

The species column from the botanist spreadsheets sometimes contains annotation suffixes encoded as dot-separated tokens (e.g. sulcata.start.rarefaction, cf.gladiata.flat.stems.1) rather than a bare species epithet. Two derived columns are created: species_clean holds the bare epithet (NA when the record only reaches genus level — i.e. when the first token is "cf", "sp", "na", or a bare number), and species_notes captures any remaining content after removing known annotation phrases. The cleaned field is used for fuzzy matching; the original species column is preserved unmodified for traceability.

Show the code 
# Recognised annotation phrases that appear as dot-joined suffixes in the
# species field but are not part of the taxon name.
annotation_pattern <- regex(
  "\\b(not in cover|start rarefaction|pink flowers|flat stems)\\b",
  ignore_case = TRUE
)

# First-token placeholders that indicate the species epithet is unresolved:
#   cf  – uncertain identification (species epithet follows as a note)
#   sp  – species unknown within the genus
#   na  – not identified
#   <number> – repeat-photo counter used as species token
unresolved_terms <- c("cf", "sp", "sp.", "na")

photo_all2 <- photo_all2 |>
  mutate(
    # Replace dot word-separators with spaces and collapse extra whitespace, so
    # "cf.gladiata.flat.stems.1" → "cf gladiata flat stems 1".
    species_raw      = str_squish(str_replace_all(coalesce(species, ""), "\\.", " ")),
    species_first    = word(species_raw, 1),
    # Species is resolved to epithet level unless the first token is a placeholder
    # term, a bare number, or a "sp" + number code (e.g. sp1, sp2 — genus-level
    # identifications where the botanist numbered multiple morphospecies within
    # the same genus rather than providing a species epithet).
    species_resolved = species_first != "" &
                       !str_to_lower(species_first) %in% unresolved_terms &
                       !str_detect(species_first, "^\\d+$") &
                       !str_detect(str_to_lower(species_first), "^sp\\d+$")
  ) |>
  mutate(
    # Bare epithet for resolved identifications; NA for genus-only records.
    species_clean = if_else(species_resolved, species_first, NA_character_),
    # Content beyond the epithet (or the whole string for unresolved records),
    # stripped of annotation phrases and bare repeat-photo counters.
    species_rest  = if_else(
      species_resolved,
      coalesce(str_squish(word(species_raw, 2, -1)), ""),
      species_raw
    ),
    species_rest  = str_squish(str_remove_all(species_rest, annotation_pattern)),
    # Remove leading or trailing bare numbers (repeat-photo counters, e.g. "1").
    species_rest  = str_squish(str_remove(species_rest, "^\\d+\\s*|\\s*\\d+$")),
    species_notes = if_else(species_rest == "", NA_character_, species_rest)
  ) |>
  select(-species_raw, -species_first, -species_resolved, -species_rest)

Join with plot-level species lists

The vegetation survey species list records species actually observed and identified during structured vegetation surveys at each plot. Survey identifications are treated as the authoritative name reference. Photo-recorded names are corrected against this list using Jaro-Winkler fuzzy string matching before any cross-dataset comparison is performed, so that minor field-recording spelling errors do not inflate mismatch counts.

Fuzzy-correct photo names against the canonical species list

Matching is performed in two sequential stages using the Jaro-Winkler string distance metric (p = 0.1 prefix weight). Stage 1 matches the photo genus against all canonical genera; a photo name that fails to find a close genus match is immediately flagged "no_match" and receives no correction. Stage 2 matches the species epithet against only the species recorded under the matched genus, constraining the candidate set to one genus. This two-stage design prevents a well-matching species epithet from pulling a poorly-matched genus across a genus boundary (e.g. “myrovernix glandulosa” would otherwise be corrected to “Rosenia glandulosa” because the epithet “glandulosa” is shared). When the genus matches but no species epithet clears its threshold, the entry is classified "genus_only" rather than forced into the nearest species. Corrections are made at the name level (not per plot), because a misspelling is a property of the entered string, not of the plot where it was recorded.

Show the code 
# Separate Jaro-Winkler thresholds for genus and species matching.
# The genus threshold is tighter — genus misspellings are typically small
# (1–2 character transpositions), so a weak genus match almost certainly
# indicates a non-taxon entry rather than a correctable spelling error.
jw_threshold_genus   <- 0.15   # genus: reject if best canonical genus is > 0.15 away
jw_threshold_species <- 0.20   # species: slightly wider to catch common misspellings

# ── Build canonical reference tables ──────────────────────────────────────────
# Split canonical names into lowercase genus and species columns for two-stage
# matching; retain the original mixed-case string for output.
canonical_df <- tibble(canonical_name = canonical_names) |>
  mutate(
    genus_l   = str_to_lower(word(canonical_name, 1)),
    species_l = str_to_lower(word(canonical_name, 2))
  )

# Unique canonical genera — target set for Stage 1 genus matching.
canonical_genera <- unique(canonical_df$genus_l)

# ── Distinct photo taxon names ─────────────────────────────────────────────────
# Use species_clean (dots removed, annotation suffixes stripped, unresolved
# epithets set to NA) so that noise in the raw species field does not inflate
# string distances during matching.
photo_names_distinct <- photo_all2 |>
  st_drop_geometry() |>
  filter(photo_type == "species", !is.na(genus), !is.na(plot)) |>
  mutate(
    photo_name = if_else(
      is.na(species_clean),
      str_trim(genus),
      paste(str_trim(genus), str_trim(species_clean))
    )
  ) |>
  distinct(photo_name) |>
  mutate(
    photo_genus_l   = str_to_lower(word(photo_name, 1)),
    # Species epithet is NA for single-word (genus-only) entries.
    photo_species_l = if_else(
      str_detect(photo_name, " "),
      str_to_lower(word(photo_name, 2)),
      NA_character_
    ),
    is_genus_only = is.na(photo_species_l)
  )

# ── Stage 1: Genus matching ────────────────────────────────────────────────────
# Match each photo genus against all canonical genera.  Only names whose best
# genus match is within jw_threshold_genus proceed to Stage 2; all others are
# immediately assigned "no_match" without any species correction applied.
# This prevents non-taxon entries like "resprouting Polygalaceous" from being
# matched — they fail at genus level before reaching species comparison at all.
genus_dist_matrix <- stringdistmatrix(
  photo_names_distinct$photo_genus_l,
  canonical_genera,
  method = "jw", p = 0.1
)

photo_names_distinct <- photo_names_distinct |>
  mutate(
    matched_genus_l = canonical_genera[apply(genus_dist_matrix, 1, which.min)],
    genus_dist      = round(apply(genus_dist_matrix, 1, min), 4),
    genus_matched   = genus_dist < jw_threshold_genus
  )

# ── Stage 2: Species matching within matched genus ─────────────────────────────
# For names that passed genus matching and carry a species epithet, match the
# epithet against only the species recorded under the matched genus in the
# canonical list.  Constraining the candidate set to one genus prevents a
# shared or similar epithet from pulling the match across a genus boundary
# (e.g. "myrovernix glandulosa" cannot reach "Rosenia glandulosa" because
# "myrovernix" fails genus matching before species comparison is attempted).
# When the genus matches but the species epithet does not clear its threshold,
# the entry is classified "genus_only" rather than forced into the nearest
# species — the epithet is treated as unresolved, not misspelled.
species_results <- photo_names_distinct |>
  filter(genus_matched, !is_genus_only) |>
  mutate(
    sp_result = map2(
      photo_species_l, matched_genus_l,
      \(sp_l, gen_l) {
        # Candidate species epithets for this matched genus (lowercase)
        candidates <- canonical_df$species_l[canonical_df$genus_l == gen_l]
        if (length(candidates) == 0 || is.na(sp_l)) {
          return(tibble(
            matched_species_l = NA_character_,
            species_dist      = NA_real_,
            species_matched   = FALSE
          ))
        }
        dists  <- stringdist(sp_l, candidates, method = "jw", p = 0.1)
        best_i <- which.min(dists)
        tibble(
          matched_species_l = candidates[best_i],
          species_dist      = round(dists[best_i], 4),
          species_matched   = dists[best_i] < jw_threshold_species
        )
      }
    )
  ) |>
  unnest(sp_result) |>
  select(photo_name, matched_species_l, species_dist, species_matched)

# Attach species results back to the full names table.
photo_names_distinct <- photo_names_distinct |>
  left_join(species_results, by = "photo_name")

# ── Assemble corrections table ─────────────────────────────────────────────────
# Join matched genus_l + species_l back to canonical_df to recover the properly
# capitalised canonical name.  Rows without a passing species match leave
# canonical_name as NA and are classified "genus_only" below.
photo_name_corrections <- photo_names_distinct |>
  left_join(
    canonical_df |> select(canonical_name, genus_l, species_l),
    by = c("matched_genus_l" = "genus_l", "matched_species_l" = "species_l")
  ) |>
  mutate(
    match_type = case_when(
      # Genus-only entry (no species epithet) that cleared the genus threshold
      is_genus_only & genus_matched                                            ~ "genus_only",
      # Genus-only entry with no genus match — non-taxon free text
      is_genus_only & !genus_matched                                           ~ "no_match",
      # Binomial where genus failed to match any canonical genus
      !genus_matched                                                           ~ "no_match",
      # Binomial where genus matched but species epithet did not clear threshold
      genus_matched & !coalesce(species_matched, FALSE)                       ~ "genus_only",
      # Both stages passed — distinguish exact from fuzzy
      genus_matched & coalesce(species_matched, FALSE) &
        str_to_lower(photo_name) == str_to_lower(canonical_name)             ~ "exact",
      genus_matched & coalesce(species_matched, FALSE)                        ~ "fuzzy",
      TRUE                                                                    ~ "no_match"
    ),
    canonical_name_matched = if_else(
      match_type %in% c("exact", "fuzzy"),
      canonical_name,
      NA_character_
    ),
    corrected_name = case_when(
      match_type %in% c("exact", "fuzzy") ~ canonical_name,
      # Recover properly capitalised genus from the canonical list
      match_type == "genus_only"          ~ str_to_title(matched_genus_l),
      TRUE                                ~ photo_name
    ),
    corrected_genus   = word(corrected_name, 1),
    corrected_species = word(corrected_name, 2),
    # Report species dist for full binomial matches, genus dist for genus-only
    match_dist = case_when(
      match_type %in% c("exact", "fuzzy") ~ species_dist,
      match_type == "genus_only"          ~ genus_dist,
      TRUE                                ~ NA_real_
    )
  ) |>
  select(photo_name, corrected_name, canonical_name_matched, match_type,
         match_dist, corrected_genus, corrected_species)

Apply corrections to the photo dataset

The correction lookup is joined back to photo_all2, adding corrected_genus, corrected_species, match_type, and match_dist columns. The corrected names are used in all subsequent joins; the original genus / species columns are preserved unmodified for traceability.

Show the code 
# Build a temporary join key in photo_all2 that matches the keys used in
# photo_names_distinct (i.e. uses species_clean, not the raw species field).
photo_all2 <- photo_all2 |>
  mutate(
    photo_name = if_else(
      photo_type == "species" & !is.na(genus) & !is.na(plot),
      if_else(
        is.na(species_clean),
        str_trim(genus),
        paste(str_trim(genus), str_trim(species_clean))
      ),
      NA_character_
    )
  ) |>
  left_join(
    select(photo_name_corrections,
           photo_name, corrected_genus, corrected_species, match_type, match_dist),
    by = "photo_name"
  ) |>
  # Fall back to the original genus/species only for rows that had no join key
  # (non-species photos).  Do NOT fall back for manually excluded entries —
  # those have match_type == "excluded" and corrected_genus intentionally NA.
  mutate(
    corrected_genus   = if_else(
      is.na(match_type) | match_type != "excluded",
      coalesce(corrected_genus, genus),
      corrected_genus
    ),
    corrected_species = if_else(
      is.na(match_type) | (match_type != "excluded" & match_type != "genus_only"),
      coalesce(corrected_species, species),
      corrected_species
    )
  ) |>
  select(-photo_name)

Rename and organize all photos

Photos were copied from the shared Google Drive source folders into a standardised local directory structure (data/photo_out/<BScpPID>/), where unmatched photos (no GPS-derived plot assignment) are placed in a 00/ subfolder. Each file was renamed according to the convention <BScpPID>_<photo_type>_<YYYYMMDD_HHMMSS>_<botanist>.jpeg, where the botanist suffix disambiguates the rare case of two photographers capturing a photo at the same plot within the same second. Files are copied byte-for-byte with file.copy() to preserve all original EXIF metadata intact; exiftool is then used to add structured provenance fields on top without re-encoding the image. This chunk is set to eval=F and must be run manually after verifying the Google Drive paths.

Show the code 
photo_all3 <- photo_all2 |>
  st_drop_geometry() |>
  mutate(
    # Use "00" as the plot folder for photos that could not be matched to a plot.
    plot_folder = if_else(is.na(BScpPID), "00", BScpPID),
    # Sanitise botanist name for use in a filename (replace spaces / special chars
    # with underscores).
    botanist_safe = str_replace_all(botanist, "[^A-Za-z0-9]", "_"),
    # Naming convention: <plot>_<type>_<YYYYMMDD_HHMMSS>_<botanist>.jpeg
    # Appending botanist prevents silent overwrites when two photographers take a
    # photo at the same plot within the same second.
    newfile = paste0(
      plot_folder, "/",
      plot_folder, "_", photo_type, "_",
      format(datetime, "%Y%m%d_%H%M%S"), "_",
      botanist_safe, ".jpeg"
    )
  )

output_folder <- "data/photo_out/"

purrr::walk(seq_len(nrow(photo_all3)), function(i) {

  input_file  <- file.path(
    "data/photos/Photos",
    sub("BioSCape_Admin/VegPlots/Photos/", "", photo_all3$folder[i]),
    photo_all3$filename[i]
  )
  output_file <- file.path(output_folder, photo_all3$newfile[i])

  if (!dir.exists(dirname(output_file)))
    dir.create(dirname(output_file), recursive = TRUE)

  input_ext <- str_to_lower(tools::file_ext(input_file))

  if (input_ext %in% c("heic", "heif")) {
    # HEIC/HEIF must be converted to JPEG — re-encoding is unavoidable.
    # After writing the pixel data, exiftool copies all original metadata
    # (GPS, camera model, exposure, etc.) from the source HEIC back into the
    # new JPEG so that provenance is preserved as completely as possible.
    image_read(input_file) |>
      image_write(output_file, format = "jpeg", quality = 95)
    # Copy original EXIF/XMP/IPTC from the HEIC source into the new JPEG.
    # The source path is wrapped in shQuote() because exiftool_call does not
    # shell-quote elements of the args list (only fnames is quoted), and paths
    # with spaces would otherwise be split by the shell into multiple tokens.
    exiftool_call(
      args = list(
        "-overwrite_original",
        paste0("-TagsFromFile=", shQuote(input_file)),
        "-all:all"
      ),
      fnames = output_file
    )
  } else {
    # For native JPEGs: copy byte-for-byte so ALL original EXIF/IPTC/XMP
    # metadata (GPS, camera model, exposure settings, colour profile, etc.)
    # is preserved without any re-encoding loss.
    file.copy(input_file, output_file, overwrite = TRUE)
  }

  # Build the corrected binomial for the Subject tag; fall back to the original
  # genus/species for plot-type photos where corrected_genus is not meaningful.
  taxon_subject <- if_else(
    photo_all3$photo_type[i] == "species" &
      !is.na(photo_all3$corrected_genus[i]),
    paste(photo_all3$corrected_genus[i], photo_all3$corrected_species[i]),
    ""
  )

  # Use the matched plot ID for the title, falling back to "unmatched" for the
  # "00" folder photos so the Title tag is still human-readable.
  plot_label <- if_else(
    is.na(photo_all3$BScpPID[i]), "unmatched", photo_all3$BScpPID[i]
  )

  # Sanitise free-text fields that may contain characters that would break the
  # exiftool command string (double-quotes, backslashes, newlines).
  sanitise <- function(x) str_replace_all(x, '["\\\\\n\r]', " ")

  # Each field is written as a separate element so that field values containing
  # spaces do not need additional quoting (exiftool treats each list element as
  # one argument).
  cmd <- list(
    "-overwrite_original",

    # XMP / IPTC provenance fields — structured, machine-readable ----------------
    # Original source path so the renamed copy is always traceable.
    paste0("-XMP:Source=",
           sanitise(paste0(photo_all3$folder[i], "/", photo_all3$filename[i]))),
    # Plot identifier as a dedicated XMP field for easy downstream filtering.
    paste0("-XMP:PlotID=", plot_label),
    # Botanist name as the EXIF Artist field (standard) and separately in XMP.
    paste0("-EXIF:Artist=",        sanitise(photo_all3$botanist[i])),
    paste0("-XMP:Creator=",        sanitise(photo_all3$botanist[i])),
    # Free-text description recorded in the field spreadsheet.
    paste0("-EXIF:ImageDescription=",
           sanitise(photo_all3$description[i])),
    # Species identification — use corrected name for Subject/Taxon tags.
    paste0("-IPTC:ObjectName=",    sanitise(taxon_subject)),
    paste0("-XMP:Subject=",        sanitise(taxon_subject)),
    # Photo type (species / plot) for easy filtering.
    paste0("-XMP:Type=",           photo_all3$photo_type[i]),
    # Title with plot and photo-type context.
    paste0("-EXIF:XPTitle=",
           sanitise(paste0("BioSCape ", photo_all3$photo_type[i],
                           " photo from plot ", plot_label))),
    # GPS coordinates from the spreadsheet as a cross-check against device GPS.
    paste0("-XMP:GPSLatitude=",    photo_all3$gps_latitude[i]),
    paste0("-XMP:GPSLongitude=",   photo_all3$gps_longitude[i]),
    # Copyright statement for the project.
    "-EXIF:Copyright=BioSCape project",
    "-XMP:Rights=BioSCape project",
    # Keywords for general discovery.
    paste0("-IPTC:Keywords=BioSCape"),
    paste0("-IPTC:Keywords=", sanitise(photo_all3$botanist[i])),
    paste0("-IPTC:Keywords=", plot_label)
  )

  exiftool_call(args = cmd, fnames = output_file)
})
